import { PassThrough } from 'node:stream';
import { Runner } from '@openai/agents';
import { isModelTimeoutError } from '../../src/flight-agent/gatewayRetryConfig.js';

function createRunner({ model, modelSettings, retry, timeoutMs }) {
  // Keep timeoutMs out of Agent.modelSettings. The request-scoped Runner is the
  // single timeout owner so each resumed attempt can use a larger value.
  return new Runner({
    model,
    modelSettings: {
      ...modelSettings,
      timeoutMs,
      retry,
    },
  });
}

function forwardText(result, output) {
  const textStream = result.toTextStream({ compatibleWithNodeStreams: true });
  let outputChars = 0;

  const textCompleted = new Promise((resolve, reject) => {
    textStream.on('data', (chunk) => {
      const text = String(chunk || '');
      outputChars += text.length;
      output.write(text);
    });
    textStream.once('end', resolve);
    textStream.once('error', reject);
  });

  return {
    get outputChars() {
      return outputChars;
    },
    async completed() {
      const settled = await Promise.allSettled([
        textCompleted,
        result.completed,
      ]);
      const failure = settled.find((item) => item.status === 'rejected');
      if (failure) throw failure.reason;
    },
  };
}

export function runStreamingWithTimeoutRecovery({
  agent,
  input,
  context,
  session,
  maxTurns,
  model,
  modelSettings,
  retry,
  initialTimeoutMs = 80_000,
  timeoutIncrementMs = 4_000,
  maxRecoveries = 4,
  onRecovery,
}) {
  const output = new PassThrough();

  const completed = (async () => {
    let runInput = input;
    let timeoutMs = initialTimeoutMs;
    let recoveries = 0;
    let outputChars = 0;

    while (true) {
      const runner = createRunner({
        model,
        modelSettings,
        retry,
        timeoutMs,
      });
      const result = await runner.run(agent, runInput, {
        context,
        session,
        maxTurns,
        stream: true,
      });
      const current = forwardText(result, output);

      try {
        await current.completed();
        outputChars += current.outputChars;
        output.end();
        return { result, recoveries, timeoutMs };
      } catch (error) {
        outputChars += current.outputChars;
        const canRecover =
          isModelTimeoutError(error) &&
          outputChars === 0 &&
          recoveries < maxRecoveries &&
          Boolean(result.state);

        if (!canRecover) throw error;

        recoveries += 1;
        const previousTimeoutMs = timeoutMs;
        timeoutMs += timeoutIncrementMs;

        await onRecovery?.({
          recovery: recoveries,
          previousTimeoutMs,
          nextTimeoutMs: timeoutMs,
        });

        // Continue the unfinished turn. Never resend the original input.
        runInput = result.state;
      }
    }
  })();

  completed.catch((error) => output.destroy(error));
  return { stream: output, completed };
}
