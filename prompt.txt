import OpenAI from "openai";
import dotenv from "dotenv";
import readline from "readline";
dotenv.config();

const client = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });

// ---- Paste your full CheapoAir System Prompt here ----
const SYSTEM_PROMPT = `
You are CheapoAir’s AI Travel Assistant.
... (your full system prompt v3) ...
`;

const rl = readline.createInterface({
  input: process.stdin,
  output: process.stdout
});

// Conversation history (system + turns)
const messages = [{ role: "system", content: SYSTEM_PROMPT }];

// JSON extractor helper
function extractJson(text) {
  if (!text) return null;
  const fenced = text.match(/```json\s*([\s\S]*?)```/i);
  const raw = fenced ? fenced[1] : text.trim();
  try { return JSON.parse(raw); } catch { return null; }
}

// Ask loop
async function ask() {
  rl.question("You: ", async (input) => {
    if (input.trim().toLowerCase() === "exit") {
      console.log("👋 Ending chat.");
      rl.close();
      return;
    }

    // Push user msg
    messages.push({ role: "user", content: input });

    try {
      const resp = await client.chat.completions.create({
        model: "gpt-4o-mini",
        temperature: 0.2,
        messages
      });

      const msg = resp.choices?.[0]?.message?.content ?? "";
      const parsed = extractJson(msg);

      if (parsed) {
        console.log("Assistant (parsed JSON):");
        console.dir(parsed, { depth: null, colors: true });
      } else {
        console.log("Assistant (raw):\n", msg);
      }

      // Save assistant msg to history
      messages.push({ role: "assistant", content: msg });
    } catch (err) {
      console.error("Error:", err);
    }

    ask(); // continue loop
  });
}

console.log("💬 CheapoAir AI Travel Assistant ready. Type 'exit' to quit.\n");
ask();
