import {
  Agent,
  retryPolicies,
} from '@openai/agents';

const transientRetryPolicy = retryPolicies.any(
  retryPolicies.providerSuggested(),
  retryPolicies.retryAfter(),
  retryPolicies.networkError(),
  retryPolicies.httpStatus([
    408,
    409,
    429,
    500,
    502,
    503,
    504,
  ]),
);

const flightAgent = new Agent({
  name: 'Flight Specialist Agent',

  // model,
  // instructions,
  // tools,

  modelSettings: {
    // Maximum time for EACH individual /v1/responses model call.
    // 1 min 20 sec = 80 sec
    timeoutMs: 80_000,

    retry: {
      // Initial request + 1 retry = max 2 attempts
      maxRetries: 1,

      backoff: {
        initialDelayMs: 500,
        maxDelayMs: 2_000,
        multiplier: 2,
        jitter: true,
      },

      policy: async (ctx) => {
        // Retry when our 80-second model timeout occurs
        if (ctx.normalized.errorCode === 'ETIMEDOUT') {
          return {
            retry: true,
            reason: `Flight model timed out on attempt ${ctx.attempt}`,
          };
        }

        // Retry normal transient OpenAI/network failures
        return transientRetryPolicy(ctx);
      },
    },
  },
});
