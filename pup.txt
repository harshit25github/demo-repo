function convertToIsoTime(dateStr, timeStr) {
  // Convert MM/DD/YYYY ➜ YYYY-MM-DD
  const [month, day, year] = dateStr.split('/');
  const formattedDate = `${year}-${month.padStart(2, '0')}-${day.padStart(2, '0')}`;

  // Combine and parse
  const fullDateTime = new Date(`${formattedDate} ${timeStr}`);

  // Format to 'YYYY-MM-DDTHH:MM:00'
  const iso = fullDateTime.toISOString().slice(0, 19); // exclude milliseconds + Z
  return iso;
}

// ✅ Example usage:
const date = '08/17/2025';
const time = '6:04 AM';
console.log(convertToIsoTime(date, time)); // ➜ '2025-08-17T06:04:00'
