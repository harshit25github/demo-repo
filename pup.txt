function getThirdWeekSaturdayOfNextMonthWithTime(timeStr) {
  const today = new Date();
  const year = today.getFullYear();
  const nextMonth = today.getMonth() + 1;

  // Step 1: Find the week that contains the 1st of next month
  const firstOfNextMonth = new Date(year, nextMonth, 1);
  const firstDayOfWeek = firstOfNextMonth.getDay(); // 0 = Sun, 1 = Mon, ...
  const offsetToSunday = -firstDayOfWeek;
  const firstWeekSunday = new Date(year, nextMonth, 1 + offsetToSunday);

  // Step 2: Get the 3rd week's Saturday (Sunday + 14 days + 6)
  const thirdWeekSaturday = new Date(firstWeekSunday);
  thirdWeekSaturday.setDate(firstWeekSunday.getDate() + 14 + 6); // 3rd week Sat

  // Step 3: Parse the time string (e.g., "1:40 AM")
  const [time, meridiem] = timeStr.trim().split(' ');
  let [hours, minutes] = time.split(':').map(Number);
  if (meridiem === 'PM' && hours !== 12) hours += 12;
  if (meridiem === 'AM' && hours === 12) hours = 0;

  // Step 4: Set time on the 3rd Saturday date
  thirdWeekSaturday.setHours(hours, minutes, 0, 0);

  // Step 5: Return in ISO format, slice off milliseconds and 'Z'
  return thirdWeekSaturday.toISOString().slice(0, 19);
}
