function getNextMonthDates() {
  const today = new Date();

  // Move to next month
  const nextMonth = today.getMonth() + 1; // JS months are 0-based
  const year = nextMonth > 11 ? today.getFullYear() + 1 : today.getFullYear();
  const month = nextMonth % 12; // wrap around if December

  // Create dates
  const date20 = new Date(year, month, 20);
  const date26 = new Date(year, month, 26);

  // Format function: MM/DD/YYYY
  const format = (date) =>
    `${(date.getMonth() + 1).toString().padStart(2, '0')}/` +
    `${date.getDate().toString().padStart(2, '0')}/` +
    `${date.getFullYear()}`;

  return {
    date20: format(date20),
    date26: format(date26),
  };
}

const { date20, date26 } = getNextMonthDates();
console.log('Next month 20th:', date20);
console.log('Next month 26th:', date26);
