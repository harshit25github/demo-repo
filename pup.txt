import { z } from "zod";

export const ComprehensiveDestinationSchema = z
  .object({
    name: z.string().describe("Destination name (city/region). Example: 'Kashmir', 'Darjeeling'."),
    country: z.string().describe("Country name where the destination is located. Example: 'India'."),
    continent: z.string().describe("Continent name. Example: 'Asia'."),
    region: z
      .string()
      .optional()
      .describe("Sub-region/state/province within the country. Example: 'West Bengal'."),
    countryCode: z
      .string()
      .describe("ISO 3166-1 alpha-2 country code. Example: 'IN'."),
    locationType: z
      .array(z.string())
      .describe(
        "Multiple tags describing what this destination is best for. Example: ['nature','mountains','relaxation']."
      ),
    overview: z
      .string()
      .describe(
        "Concise-but-rich overview: what it is, why it’s popular, what kind of trip it suits."
      ),

    attractions: z
      .array(
        z.object({
          name: z.string().describe("Attraction/POI name."),
          description: z
            .string()
            .describe("What it is + why visit + what to do there (practical + experiential)."),
          hours: z
            .string()
            .optional()
            .describe("Typical visiting hours or best visiting window. Use 'Varies' if uncertain."),
          entryFee: z
            .string()
            .optional()
            .describe("Typical entry fee info. Use 'Varies' / 'Check official source' if uncertain."),
          tips: z
            .string()
            .optional()
            .describe("High-signal tips: crowds, tickets, weather, photo spots, local hacks."),
        })
      )
      .describe("Key attractions and experiences inside the destination."),

    logistics: z
      .object({
        bestTimeToVisit: z
          .string()
          .describe("Best season/months + why (weather, crowd levels, activities)."),
        transportTips: z
          .string()
          .describe("How to get in/out + how to get around locally (modes, typical constraints)."),
        accommodationSuggestions: z
          .array(z.string())
          .optional()
          .describe("Recommended areas/neighborhoods or stay styles (houseboat, hill-view hotel, etc.)."),
      })
      .describe("Operational travel planning information for realistic itineraries."),

    culturalTips: z
      .string()
      .optional()
      .describe("Local etiquette, norms, sensitive topics, dress guidance, permits, etc."),

    localEvents: z
      .array(z.string())
      .optional()
      .describe("Notable festivals/events (or seasonal highlights) that can shape an itinerary."),

    nearbyPlaces: z
      .array(z.string())
      .optional()
      .describe("Good nearby add-ons / day-trips / route-linking places."),
  })
  .describe(
    "Comprehensive destination knowledge object used to ground an itinerary generator with reliable, structured travel facts."
  );
