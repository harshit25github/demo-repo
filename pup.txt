import {
  ModelTimeoutError,
  retryPolicies,
} from '@openai/agents';

policy: async (ctx) => {
  console.log('[Flight Retry Debug]', {
    attempt: ctx.attempt,
    maxRetries: ctx.maxRetries,

    errorName:
      ctx.error instanceof Error
        ? ctx.error.name
        : undefined,

    errorMessage:
      ctx.error instanceof Error
        ? ctx.error.message
        : undefined,

    errorCode: ctx.normalized?.errorCode,

    stream: ctx.stream,
    responseStarted: ctx.responseStarted,
    replaySafety: ctx.replaySafety,
    statefulRequest: ctx.statefulRequest,

    providerAdvice: ctx.providerAdvice,
  });

  const isTimeout =
    ctx.error instanceof ModelTimeoutError ||
    ctx.normalized?.errorCode === 'ETIMEDOUT';

  if (isTimeout) {
    console.log('[Flight Retry] Timeout detected → retry requested');

    return {
      retry: true,
      reason: 'Flight model timed out after 80 seconds',
    };
  }

  return transientRetryPolicy(ctx);
},
