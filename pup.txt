function getThirdWeekDayTimesCSV(timeArray, dayType = 'sunday') {
  const today = new Date();
  const year = today.getFullYear();
  const nextMonth = today.getMonth() + 1;

  // 1. Get the 1st of next month
  const firstOfNextMonth = new Date(year, nextMonth, 1);

  // 2. Find the correct weekday offset
  const targetDay = dayType.toLowerCase() === 'saturday' ? 6 : 0; // 0 = Sunday, 6 = Saturday
  const firstDayOfMonth = firstOfNextMonth.getDay(); // 0 = Sun, 1 = Mon, ..., 6 = Sat
  let dayOffset = (7 + targetDay - firstDayOfMonth) % 7;

  // 3. Get the 1st target day (saturday/sunday), then add 14 days for 3rd week
  const thirdWeekTargetDate = new Date(year, nextMonth, 1 + dayOffset + 14);

  // 4. Helper for formatting
  const pad = (n) => String(n).padStart(2, '0');

  // 5. Map over each time string
  const result = timeArray.map((timeStr) => {
    const [time, meridiem] = timeStr.trim().split(' ');
    let [hours, minutes] = time.split(':').map(Number);
    if (meridiem === 'PM' && hours !== 12) hours += 12;
    if (meridiem === 'AM' && hours === 12) hours = 0;

    const dateWithTime = new Date(thirdWeekTargetDate);
    dateWithTime.setHours(hours, minutes, 0, 0);
    dateWithTime.setHours(dateWithTime.getHours() - 4); // Subtract 4 hours

    return `${dateWithTime.getFullYear()}-${pad(dateWithTime.getMonth() + 1)}-${pad(dateWithTime.getDate())}T${pad(dateWithTime.getHours())}:${pad(dateWithTime.getMinutes())}:00`;
  });

  return result.join(', ');
}
