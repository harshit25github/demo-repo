function getThirdWeekSunSatOfNextMonth() {
  const today = new Date();
  const year = today.getFullYear();
  const nextMonth = today.getMonth() + 1;

  // Find the Sunday of the week containing the 1st of the month
  const firstOfNextMonth = new Date(year, nextMonth, 1);
  const firstDayOfWeek = firstOfNextMonth.getDay(); // 0 = Sun, 1 = Mon...
  const offsetToSunday = -firstDayOfWeek; // can be negative or 0
  const firstWeekSunday = new Date(year, nextMonth, 1 + offsetToSunday);

  // 3rd week's Sunday = first week Sunday + 14 days
  const thirdWeekSunday = new Date(firstWeekSunday);
  thirdWeekSunday.setDate(firstWeekSunday.getDate() + 14);

  const thirdWeekSaturday = new Date(thirdWeekSunday);
  thirdWeekSaturday.setDate(thirdWeekSunday.getDate() + 6);

  const format = (date) => {
    const mm = String(date.getMonth() + 1).padStart(2, '0');
    const dd = String(date.getDate()).padStart(2, '0');
    const yyyy = date.getFullYear();
    return `${mm}/${dd}/${yyyy}`;
  };

  return {
    sunday: format(thirdWeekSunday),
    saturday: format(thirdWeekSaturday),
  };
}
