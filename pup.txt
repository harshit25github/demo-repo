async function waitForXPathText(page, text, xpath, timeout = 10000, interval = 500) {
  const maxTime = Date.now() + timeout;

  while (Date.now() < maxTime) {
    const elements = await page.$$(xpath);
    for (const el of elements) {
      const elText = await page.evaluate(el => el.textContent.trim(), el);
      if (elText === text) return true;
    }

    await new Promise(res => setTimeout(res, interval));
  }

  throw new Error(`XPath element with text "${text}" not found within ${timeout}ms`);
}
