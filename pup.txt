const axios = require("axios");

async function getCityCode(city) {
  try {
    if (!city || typeof city !== "string" || !city.trim()) {
      return {
        status: false,
        message: "City is required",
      };
    }

    const url = `https://www.cheapoair.com/airautosuggestapi/autosuggest/v2/en-us/${encodeURIComponent(
      city.trim()
    )}/true/50`;

    const { data } = await axios.get(url, {
      headers: {
        Accept: "application/json, text/plain, */*",
        "x-client-id": "landingpage",
        Referer: "https://www.cheapoair.com/",
        "User-Agent":
          "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",
      },
    });

    const locations = data?.Location || [];

    if (!locations.length) {
      return {
        status: false,
        message: "City not found",
      };
    }

    // Prefer city result, fallback to first available result
    const selectedLocation =
      locations.find((item) => item.Type?.toLowerCase() === "city") ||
      locations[0];

    return {
      status: true,
      code: selectedLocation.Code,
      name: selectedLocation.Text,
    };
  } catch (error) {
    return {
      status: false,
      message: "Failed to fetch city details",
    };
  }
}

module.exports = {
  getCityCode,
};
