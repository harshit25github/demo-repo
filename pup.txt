function getCalendarPositionFromDate(dateString) {
  const [mm, dd, yyyy] = dateString.split("-").map(Number);
  const targetDate = new Date(yyyy, mm - 1, dd);

  // 1st day of the month
  const firstDay = new Date(yyyy, mm - 1, 1);
  const startDayOfWeek = firstDay.getDay(); // 0 = Sunday, 6 = Saturday

  // Calculate day offset from the first day
  const dayOffset = dd + startDayOfWeek - 1;

  const tr = Math.floor(dayOffset / 7) + 1; // Row number (week)
  const td = (dayOffset % 7) + 1;           // Column number (day of week)

  return { tr, td };
}
