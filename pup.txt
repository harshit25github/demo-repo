import OpenAI from "openai";

const client = new OpenAI({
  apiKey: process.env.OPENAI_API_KEY,
  organization: process.env.OPENAI_ORG_ID,
  project: process.env.OPENAI_PROJECT_ID,
});

async function getAccessibleFineTunedModels(): Promise<string[]> {
  const modelIds = new Set<string>();

  const models = await client.models.list();

  for (const model of models.data) {
    if (model.id.startsWith("ft:")) {
      modelIds.add(model.id);
    }
  }

  for await (const job of client.fineTuning.jobs.list({ limit: 100 })) {
    if (job.status === "succeeded" && job.fine_tuned_model) {
      modelIds.add(job.fine_tuned_model);
    }
  }

  return [...modelIds].sort();
}

const fineTunedModels = await getAccessibleFineTunedModels();

console.log(fineTunedModels);
