policy: async (ctx) => {
  console.log('MODEL RETRY CONTEXT', {
    attempt: ctx.attempt,
    maxRetries: ctx.maxRetries,

    stream: ctx.stream,
    responseStarted: ctx.responseStarted,
    replaySafety: ctx.replaySafety,
    statefulRequest: ctx.statefulRequest,

    errorCode: ctx.normalized.errorCode,
    isAbort: ctx.normalized.isAbort,

    errorName:
      ctx.error instanceof Error
        ? ctx.error.constructor.name
        : undefined,

    providerAdvice: ctx.providerAdvice,
  });

  if (ctx.normalized.errorCode === 'ETIMEDOUT') {
    console.log('TIMEOUT → requesting retry');

    return {
      retry: true,
      reason: 'Flight model timed out',
    };
  }

  return transientRetryPolicy(ctx);
}
