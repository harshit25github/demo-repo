async function waitForElementWithText(page, text, selector = '*', timeout = 10000, interval = 500) {
  const maxTime = Date.now() + timeout;

  while (Date.now() < maxTime) {
    const found = await page.$$eval(selector, (elements, targetText) => {
      return elements.some(el => el.innerText.trim() === targetText);
    }, text);

    if (found) return true;

    await new Promise(res => setTimeout(res, interval));
  }

  throw new Error(`Element with text "${text}" not found within ${timeout}ms`);
}
